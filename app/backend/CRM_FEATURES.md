## 4. Database Schema

### 4.1 Core Tables

```sql
-- CRM Contacts (extends users table)
CREATE TABLE crm_contacts (
    id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT NOT NULL,
    assigned_advisor_id INT DEFAULT NULL,
    assigned_support_agent_id INT DEFAULT NULL,
    tier ENUM('standard', 'premium', 'vip') DEFAULT 'standard',
    risk_profile ENUM('conservative', 'moderate', 'aggressive') DEFAULT 'moderate',
    lifetime_value DECIMAL(15,2) DEFAULT 0.00,
    acquisition_source VARCHAR(100),
    tags JSON,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    FOREIGN KEY (assigned_advisor_id) REFERENCES users(id) ON DELETE SET NULL,
    FOREIGN KEY (assigned_support_agent_id) REFERENCES users(id) ON DELETE SET NULL,
    INDEX idx_assigned_advisor (assigned_advisor_id),
    INDEX idx_assigned_agent (assigned_support_agent_id),
    INDEX idx_tier (tier)
);

-- CRM Activities (calls, emails, meetings, notes)
CREATE TABLE crm_activities (
    id INT PRIMARY KEY AUTO_INCREMENT,
    contact_id INT NOT NULL,
    user_id INT NOT NULL, -- advisor or agent performing activity
    activity_type ENUM('call', 'email', 'meeting', 'note', 'sms', 'trade_review', 'system') NOT NULL,
    subject VARCHAR(255),
    description TEXT,
    outcome VARCHAR(100),
    duration INT DEFAULT NULL, -- in seconds (for calls/meetings)
    scheduled_at TIMESTAMP DEFAULT NULL,
    completed_at TIMESTAMP DEFAULT NULL,
    recording_url VARCHAR(500) DEFAULT NULL,
    metadata JSON, -- additional data (call_id, email_id, etc.)
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (contact_id) REFERENCES crm_contacts(id) ON DELETE CASCADE,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    INDEX idx_contact_id (contact_id),
    INDEX idx_activity_type (activity_type),
    INDEX idx_scheduled_at (scheduled_at),
    INDEX idx_completed_at (completed_at)
);

-- CRM Tickets (customer support)
CREATE TABLE crm_tickets (
    id INT PRIMARY KEY AUTO_INCREMENT,
    ticket_number VARCHAR(50) UNIQUE NOT NULL,
    contact_id INT NOT NULL,
    assigned_agent_id INT DEFAULT NULL,
    category ENUM('account', 'deposit', 'withdrawal', 'trading', 'technical', 'kyc', 'other') NOT NULL,
    priority ENUM('low', 'medium', 'high', 'urgent') DEFAULT 'medium',
    status ENUM('new', 'open', 'pending', 'resolved', 'closed') DEFAULT 'new',
    subject VARCHAR(255) NOT NULL,
    description TEXT NOT NULL,
    resolution TEXT DEFAULT NULL,
    source ENUM('email', 'phone', 'chat', 'web', 'social') DEFAULT 'web',
    sla_breach BOOLEAN DEFAULT FALSE,
    first_response_at TIMESTAMP DEFAULT NULL,
    resolved_at TIMESTAMP DEFAULT NULL,
    closed_at TIMESTAMP DEFAULT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (contact_id) REFERENCES crm_contacts(id) ON DELETE CASCADE,
    FOREIGN KEY (assigned_agent_id) REFERENCES users(id) ON DELETE SET NULL,
    INDEX idx_ticket_number (ticket_number),
    INDEX idx_contact_id (contact_id),
    INDEX idx_assigned_agent (assigned_agent_id),
    INDEX idx_status (status),
    INDEX idx_priority (priority),
    INDEX idx_created_at (created_at)
);

-- CRM Tasks (follow-ups, reminders)
CREATE TABLE crm_tasks (
    id INT PRIMARY KEY AUTO_INCREMENT,
    contact_id INT NOT NULL,
    assigned_to INT NOT NULL,
    task_type ENUM('call', 'email', 'meeting', 'review', 'follow_up', 'other') NOT NULL,
    title VARCHAR(255) NOT NULL,
    description TEXT,
    priority ENUM('low', 'medium', 'high') DEFAULT 'medium',
    status ENUM('pending', 'in_progress', 'completed', 'cancelled') DEFAULT 'pending',
    due_date TIMESTAMP NOT NULL,
    completed_at TIMESTAMP DEFAULT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (contact_id) REFERENCES crm_contacts(id) ON DELETE CASCADE,
    FOREIGN KEY (assigned_to) REFERENCES users(id) ON DELETE CASCADE,
    INDEX idx_contact_id (contact_id),
    INDEX idx_assigned_to (assigned_to),
    INDEX idx_status (status),
    INDEX idx_due_date (due_date)
);

-- CRM Calls (softphone call logs)
CREATE TABLE crm_calls (
    id INT PRIMARY KEY AUTO_INCREMENT,
    contact_id INT NOT NULL,
    user_id INT NOT NULL, -- advisor or agent
    call_type ENUM('inbound', 'outbound') NOT NULL,
    phone_number VARCHAR(20) NOT NULL,
    duration INT DEFAULT 0, -- in seconds
    status ENUM('completed', 'missed', 'voicemail', 'failed') NOT NULL,
    recording_url VARCHAR(500) DEFAULT NULL,
    voice_effect VARCHAR(50) DEFAULT 'natural', -- natural, robot, pitch_shift, etc.
    notes TEXT,
    call_metadata JSON, -- peer connection info, quality metrics
    started_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    ended_at TIMESTAMP DEFAULT NULL,
    FOREIGN KEY (contact_id) REFERENCES crm_contacts(id) ON DELETE CASCADE,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    INDEX idx_contact_id (contact_id),
    INDEX idx_user_id (user_id),
    INDEX idx_started_at (started_at),
    INDEX idx_status (status)
);

-- CRM Campaigns (email/SMS campaigns)
CREATE TABLE crm_campaigns (
    id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(255) NOT NULL,
    campaign_type ENUM('email', 'sms', 'push') NOT NULL,
    target_audience JSON, -- filters: tier, activity, balance, etc.
    subject VARCHAR(255),
    content TEXT NOT NULL,
    status ENUM('draft', 'scheduled', 'running', 'completed', 'paused') DEFAULT 'draft',
    scheduled_at TIMESTAMP DEFAULT NULL,
    completed_at TIMESTAMP DEFAULT NULL,
    sent_count INT DEFAULT 0,
    opened_count INT DEFAULT 0,
    clicked_count INT DEFAULT 0,
    created_by INT NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (created_by) REFERENCES users(id) ON DELETE CASCADE,
    INDEX idx_status (status),
    INDEX idx_campaign_type (campaign_type)
);

-- CRM Performance (advisor/agent metrics)
CREATE TABLE crm_performance (
    id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT NOT NULL,
    period_start DATE NOT NULL,
    period_end DATE NOT NULL,
    role ENUM('advisor', 'support') NOT NULL,
    -- Advisor metrics
    clients_acquired INT DEFAULT 0,
    revenue_generated DECIMAL(15,2) DEFAULT 0.00,
    commission_earned DECIMAL(15,2) DEFAULT 0.00,
    client_retention_rate DECIMAL(5,2) DEFAULT 0.00,
    -- Support metrics
    tickets_resolved INT DEFAULT 0,
    avg_response_time INT DEFAULT 0, -- in seconds
    avg_resolution_time INT DEFAULT 0, -- in seconds
    customer_satisfaction DECIMAL(3,2) DEFAULT 0.00, -- 0.00 to 5.00
    sla_compliance_rate DECIMAL(5,2) DEFAULT 0.00,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    INDEX idx_user_id (user_id),
    INDEX idx_period (period_start, period_end),
    INDEX idx_role (role)
);

-- Voice Preferences (per user)
CREATE TABLE crm_voice_preferences (
    id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT NOT NULL,
    voice_effect VARCHAR(50) DEFAULT 'natural',
    pitch_shift INT DEFAULT 0, -- -12 to +12 semitones
    use_voice_changer BOOLEAN DEFAULT FALSE,
    noise_cancellation BOOLEAN DEFAULT TRUE,
    auto_record BOOLEAN DEFAULT TRUE,
    preferences_json JSON,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    UNIQUE KEY unique_user (user_id)
);
```

### 6.1 Call Signaling
```javascript
// Initiate call
{
    type: 'call_initiate',
    to_user_id: 123,
    contact_id: 456,
    call_type: 'voice' | 'video'
}

// Call offer (WebRTC SDP)
{
    type: 'call_offer',
    call_id: 'uuid',
    sdp: 'v=0\no=...',
    ice_candidates: []
}

// Call answer
{
    type: 'call_answer',
    call_id: 'uuid',
    sdp: 'v=0\no=...'
}

// Call ended
{
    type: 'call_ended',
    call_id: 'uuid',
    duration: 180,
    reason: 'completed' | 'rejected' | 'missed'
}

// Voice effect change (real-time)
{
    type: 'voice_effect_change',
    call_id: 'uuid',
    effect: 'natural' | 'robot' | 'pitch_shift',
    parameters: { pitch: +5 }
}
```
